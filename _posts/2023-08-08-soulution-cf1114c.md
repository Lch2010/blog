[题目传送门](https://www.luogu.com.cn/problem/CF1114C)

#### 题目大意

计算 $$n$$ 的阶乘在 $$k$$ 进制下末尾 $$0$$ 的个数。

#### 思路

在 $$10$$ 进制下，我们只需要记录 $$2$$，$$5$$ 的个数取 $$\min$$ 即可。那么我们可以推知，当进制为 $$k$$ 时，只要对 $$k$$ 进行质因数分解即可，然后取每个质数，数有多少个，并除以这个质数在 $$k$$ 中出现的次数，取一个当中最小值 $$\min$$ 即可。

本质：$$k$$ 进制下是满 $$k$$ 进位。

#### 注意事项

`for(int l=k;l<=n;l*=k)` 可能会爆 `long long`，可以开一个 `__int128`。

#### AC Code

```cpp
#include<bits/stdc++.h>
#define int __int128
using namespace std;
int ans=1e18;
inline int read(){
	int s = 0, w = 1; char ch =   getchar();
	while(ch < '0' || ch > '9'){ if(ch == '-') w = -1; ch = getchar(); }
    while(ch >= '0' && ch <= '9') s = s * 10 + ch - '0', ch = getchar();
    return s*w;
}
void write(int x) {
    if(x<0) putchar('-'),x=-x;
    if(x>9) write(x/10);
    putchar(x%10+'0');
}
signed main(){
// 	freopen("zero.in","r",stdin);
// 	freopen("zero.out","w",stdout);
	ios::sync_with_stdio(false);
	cin.tie(0);cout.tie(0);
	int n,k;
	n=read(),k=read();
	int cnt2=0,cnt5=0;
	for(int i=2;i<=k/i;i++){
		int j=0;
		while(k%i==0)	j++,k/=i;
		if(j){
			int Ans=0;
			for(int l=i;l<=n;l*=i){
				Ans+=n/l;
			}
			Ans/=j;
			ans=min(ans,Ans);
		}
	}
	if(k>1){
		int Ans=0;
		for(int l=k;l<=n;l*=k){
			Ans+=n/l;
		}
		ans=min(ans,Ans);
	}
	write(ans);	
	return 0;
}
```
